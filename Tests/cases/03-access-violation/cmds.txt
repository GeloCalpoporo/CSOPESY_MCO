# MCO2 spec sections 5 and 7.
# A 64-byte process owns nothing but its symbol table, so 0x500 is far outside
# its address space. Writing there must kill it like a segfault, and screen -r
# must then report how and when it died.
initialize
screen -c segfault 64 "DECLARE ok 1; WRITE 0x500 42; PRINT(\"never reached\")"
@sleep 1
exit
screen -r segfault
screen -r ghost
screen -ls
exit
