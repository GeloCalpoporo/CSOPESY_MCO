# MCO2 spec section 4: memory must be a power of 2 in [2^6, 2^16] bytes.
# Anything else prints exactly "invalid memory allocation".
initialize
# --- legal: the two boundaries ---
screen -s okmin 64
exit
screen -s okmax 65536
exit
# --- illegal: not a power of 2 ---
screen -s bad100 100
# --- illegal: below 2^6 ---
screen -s bad32 32
# --- illegal: above 2^16 ---
screen -s bad131072 131072
# --- illegal: not a number at all ---
screen -s badword abc
screen -ls
exit
