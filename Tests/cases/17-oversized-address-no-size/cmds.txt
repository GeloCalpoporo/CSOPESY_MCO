# Two things at once:
#  1. "screen -c" is given NO memory size - the size must fall back to
#     max-mem-per-proc rather than being rejected as a syntax error.
#  2. That makes the process 256 bytes, so 0x500 is far outside it. The write
#     must be an access violation, and the program must not hang trying to
#     service an instruction that a single frame can never satisfy.
initialize
screen -c faulty_process "DECLARE varA 10; DECLARE varB 5; ADD varA varA varB; WRITE 0x500 varA; READ varC 0x500; PRINT(\"Variable A: \" + varA); PRINT(\"Result: \" + varC)"
@sleep 2
exit
screen -ls
screen -r faulty_process
exit
